## Relay生成配对文件

DJ_MODULE_ID='dji-75d81d083dca'
DJ_ENROLL_FILE="enrollment-$(date +%Y%m%d-%H%M%S).json"

  --state-dir /var/lib/djonehub-relay --tenant home \

  --out "/var/lib/djonehub-relay/transfer/$DJ_ENROLL_FILE"
